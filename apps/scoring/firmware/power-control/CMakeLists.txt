cmake_minimum_required(VERSION 3.25)
project(scoring_power_control LANGUAGES C)
enable_testing()
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/ScoringNativePolicies.cmake")
scoring_native_initialize()
option(SCORING_ENABLE_LLVM_COVERAGE "Enable native LLVM coverage" OFF)
if(SCORING_ENABLE_LLVM_COVERAGE)
  scoring_native_enable_llvm_coverage()
endif()
foreach(suite policy virtual_policy target)
  add_executable(scoring_power_${suite}_tests tests/test_${suite}.c power_control.c)
  target_include_directories(scoring_power_${suite}_tests PRIVATE .)
  scoring_native_apply_baseline_warnings(scoring_power_${suite}_tests)
  add_test(NAME power_${suite} COMMAND scoring_power_${suite}_tests)
endforeach()
target_sources(scoring_power_target_tests PRIVATE stm32c011_power.c)
target_compile_definitions(scoring_power_target_tests PRIVATE POWER_REGISTER_TEST)
